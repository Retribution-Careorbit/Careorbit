# STEP 2 — POPULATE KEY VAULT WITH ALL SECRETS (5 minutes)

Run this after Step 1 completes. It extracts every key from every service and stores it in Key Vault.

```bash
#!/bin/bash
set -euo pipefail

RG="careorbit-production"
KV="careorbit-kv-prod"
PG_PASS="{{PASTE_PROD_PG_PASSWORD_FROM_STEP_1}}"

echo "→ Collecting all service credentials..."

# PostgreSQL
az keyvault secret set --vault-name $KV --name DATABASE-URL \
  --value "postgresql://careorbit_admin:${PG_PASS}@careorbit-db-prod.postgres.database.azure.com:5432/careorbit?sslmode=require" -o none

# OpenAI
OPENAI_EP=$(az cognitiveservices account show --name careorbit-openai --resource-group $RG --query "properties.endpoint" -o tsv)
OPENAI_KEY=$(az cognitiveservices account keys list --name careorbit-openai --resource-group $RG --query "key1" -o tsv)
az keyvault secret set --vault-name $KV --name AZURE-OPENAI-ENDPOINT --value "$OPENAI_EP" -o none
az keyvault secret set --vault-name $KV --name AZURE-OPENAI-KEY --value "$OPENAI_KEY" -o none

# Document Intelligence
DI_EP=$(az cognitiveservices account show --name careorbit-docai --resource-group $RG --query "properties.endpoint" -o tsv)
DI_KEY=$(az cognitiveservices account keys list --name careorbit-docai --resource-group $RG --query "key1" -o tsv)
az keyvault secret set --vault-name $KV --name AZURE-DI-ENDPOINT --value "$DI_EP" -o none
az keyvault secret set --vault-name $KV --name AZURE-DI-KEY --value "$DI_KEY" -o none

# Language NER
LANG_EP=$(az cognitiveservices account show --name careorbit-language --resource-group $RG --query "properties.endpoint" -o tsv)
LANG_KEY=$(az cognitiveservices account keys list --name careorbit-language --resource-group $RG --query "key1" -o tsv)
az keyvault secret set --vault-name $KV --name AZURE-LANGUAGE-ENDPOINT --value "$LANG_EP" -o none
az keyvault secret set --vault-name $KV --name AZURE-LANGUAGE-KEY --value "$LANG_KEY" -o none

# AI Search
SEARCH_KEY=$(az search admin-key show --service-name careorbit-search --resource-group $RG --query "primaryKey" -o tsv)
az keyvault secret set --vault-name $KV --name AZURE-SEARCH-ENDPOINT --value "https://careorbit-search.search.windows.net" -o none
az keyvault secret set --vault-name $KV --name AZURE-SEARCH-KEY --value "$SEARCH_KEY" -o none
az keyvault secret set --vault-name $KV --name AZURE-SEARCH-INDEX --value "drug-interactions" -o none

# Translator
TRANS_EP=$(az cognitiveservices account show --name careorbit-translator --resource-group $RG --query "properties.endpoint" -o tsv)
TRANS_KEY=$(az cognitiveservices account keys list --name careorbit-translator --resource-group $RG --query "key1" -o tsv)
az keyvault secret set --vault-name $KV --name AZURE-TRANSLATOR-ENDPOINT --value "$TRANS_EP" -o none
az keyvault secret set --vault-name $KV --name AZURE-TRANSLATOR-KEY --value "$TRANS_KEY" -o none

# Communication Services
COMM_STR=$(az communication list-key --name careorbit-comm --resource-group $RG --query "primaryConnectionString" -o tsv)
az keyvault secret set --vault-name $KV --name AZURE-COMM-CONNECTION-STRING --value "$COMM_STR" -o none

# Blob Storage
BLOB_STR=$(az storage account show-connection-string --name careorbitprod --resource-group $RG --query "connectionString" -o tsv)
az keyvault secret set --vault-name $KV --name AZURE-BLOB-CONNECTION-STRING --value "$BLOB_STR" -o none

# Application Insights
INSIGHTS_STR=$(az monitor app-insights component show --app careorbit-insights-prod --resource-group $RG --query "connectionString" -o tsv)
az keyvault secret set --vault-name $KV --name APPINSIGHTS-CONNECTION-STRING --value "$INSIGHTS_STR" -o none

# Application secrets (generate fresh)
JWT_SECRET=$(openssl rand -hex 32)
PG_ENC_KEY=$(openssl rand -hex 32)
az keyvault secret set --vault-name $KV --name JWT-SECRET-KEY --value "$JWT_SECRET" -o none
az keyvault secret set --vault-name $KV --name PG-ENCRYPTION-KEY --value "$PG_ENC_KEY" -o none

echo ""
echo "══════════════════════════════════════════"
echo "ALL 17 SECRETS STORED IN KEY VAULT"
echo "══════════════════════════════════════════"
az keyvault secret list --vault-name $KV --query "[].name" -o table
```

**Verify:** That last command should show 17 secrets.

## 6.2 GitHub Actions Workflow

**File: `.github/workflows/deploy.yml`**

```yaml
name: CareOrbit CI/CD

on:
  push:
    branches: [main, release, dev]
  pull_request:
    branches: [main]

env:
  PYTHON_VERSION: "3.12"
  NODE_VERSION: "20"

jobs:
  # ═══ TEST (runs on every push/PR) ═══
  test:
    runs-on: ubuntu-latest
    services:
      postgres:
        image: postgres:16
        env:
          POSTGRES_USER: test
          POSTGRES_PASSWORD: test
          POSTGRES_DB: careorbit_test
        ports: ["5432:5432"]
        options: --health-cmd pg_isready --health-interval 10s --health-timeout 5s --health-retries 5
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with: { python-version: "${{ env.PYTHON_VERSION }}" }
      - run: pip install -r apps/backend/requirements.txt -r apps/backend/requirements-test.txt
      - name: Run 875-test TDD suite
        working-directory: apps/backend
        env:
          DATABASE_URL: postgresql://test:test@localhost:5432/careorbit_test
          JWT_SECRET_KEY: test-ci-key
          PG_ENCRYPTION_KEY: test-ci-enc
          ENVIRONMENT: test
        run: pytest tests/ -v --tb=short --junitxml=test-results.xml
      - uses: actions/upload-artifact@v4
        if: always()
        with: { name: test-results, path: apps/backend/test-results.xml }

  # ═══ DEPLOY DEV (on push to dev branch) ═══
  deploy-dev:
    needs: test
    if: github.ref == 'refs/heads/dev' && github.event_name == 'push'
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: azure/login@v2
        with: { creds: "${{ secrets.AZURE_CREDENTIALS }}" }
      - uses: azure/webapps-deploy@v3
        with: { app-name: careorbit-api-dev, package: apps/backend }

  # ═══ DEPLOY RELEASE (on push to release branch) ═══
  deploy-release:
    needs: test
    if: github.ref == 'refs/heads/release' && github.event_name == 'push'
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: azure/login@v2
        with: { creds: "${{ secrets.AZURE_CREDENTIALS }}" }
      - uses: azure/webapps-deploy@v3
        with: { app-name: careorbit-api-rel, package: apps/backend }

  # ═══ DEPLOY PRODUCTION (on push to main — via staging slot) ═══
  deploy-production:
    needs: test
    if: github.ref == 'refs/heads/main' && github.event_name == 'push'
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: azure/login@v2
        with: { creds: "${{ secrets.AZURE_CREDENTIALS }}" }

      # Deploy to staging slot first
      - uses: azure/webapps-deploy@v3
        with:
          app-name: careorbit-api-prod
          slot-name: staging
          package: apps/backend

      # Swap staging → production (zero downtime)
      - name: Swap to production
        run: |
          az webapp deployment slot swap \
            --name careorbit-api-prod \
            --resource-group careorbit-production \
            --slot staging \
            --target-slot production

  # ═══ DEPLOY FRONTEND (on push to main) ═══
  deploy-frontend:
    needs: test
    if: github.ref == 'refs/heads/main' && github.event_name == 'push'
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: { node-version: "${{ env.NODE_VERSION }}" }
      - working-directory: apps/frontend
        run: npm ci && npm run build
        env:
          VITE_API_BASE_URL: https://careorbit-api-prod.azurewebsites.net
      - uses: Azure/static-web-apps-deploy@v1
        with:
          azure_static_web_apps_api_token: ${{ secrets.SWA_DEPLOYMENT_TOKEN }}
          repo_token: ${{ secrets.GITHUB_TOKEN }}
          action: upload
          app_location: apps/frontend/dist
          skip_app_build: true
```

# STEP 9 — DATABASE MIGRATION FROM REPLIT (5 minutes)

```bash
# In Replit shell — export your current database
pg_dump $DATABASE_URL --schema-only > schema.sql
pg_dump $DATABASE_URL --data-only > data.sql

# Import to all three environments
for ENV in prod dev rel; do
  PG_URL=$(az keyvault secret show --vault-name careorbit-kv-prod --name DATABASE-URL --query value -o tsv)
  # For dev/rel, construct URL manually from passwords saved in Step 1
  
  psql "$PG_URL" -c "CREATE EXTENSION IF NOT EXISTS pgcrypto;"
  psql "$PG_URL" < schema.sql
  psql "$PG_URL" < data.sql
  echo "✅ Migrated to $ENV"
done
```

# ROLLBACK CHEAT SHEET

| Problem | Fix | Time |
|---|---|---|
| Bad production deploy | `az webapp deployment slot swap --name careorbit-api-prod --resource-group careorbit-production --slot production --target-slot staging` | 30 sec |
| Bad frontend deploy | Revert commit on main, push → auto-redeploys | 2 min |
| Database corruption | `az postgres flexible-server restore --name careorbit-db-restored --source-server careorbit-db-prod --restore-time "{{ISO_TIMESTAMP}}"` | 15 min |
| Replit can't connect | `curl -s ifconfig.me` → update firewall rule with new IP | 1 min |
| Key Vault locked | Check RBAC: `az role assignment list --scope $(az keyvault show --name careorbit-kv-prod --query id -o tsv)` | 2 min |
